def test_source_guard_follows_domain_passthrough_and_safe_fallbacks(self):
    manifest = json.loads((ROOT / 'tests/contracts/node-http-error-boundaries.json').read_text())
    observed = {'unauthorized', 'source_unavailable'}
    for path in [*(SOURCE / 'skills/sources').glob('*.py'), SOURCE / 'skills/interface/sources/handler.py']:
        for node in ast.walk(ast.parse(path.read_text())):
            if not (isinstance(node, ast.Raise) and isinstance(node.exc, ast.Call) and node.exc.args):
                continue
            category = getattr(node.exc.func, 'id', getattr(node.exc.func, 'attr', ''))
            argument = node.exc.args[0]
            if category == 'SourceError':
                self.assertIsInstance(argument, ast.Constant, str(path))
                self.assertIsInstance(argument.value, str, str(path))
            if category == 'SourceError' and isinstance(argument, ast.Constant) and isinstance(argument.value, str):
                observed.add(argument.value)
    self.assertEqual(observed, set(manifest['http_boundary_codes']['source_handler']))
    self.assertEqual(manifest['internal_only']['operation_handler'], ['skill_request_invalid'])

def test_loopback_service_auth_and_error_redaction(self):
    import threading
    from urllib import request
    from urllib import error
    from laomedo.github.interface.issues.server import IssueServer as server
    token = 'dummy-control-' + 'x' * 40
    service = server(self.gateway, token, port=0)
    thread = threading.Thread(target=service.serve_forever, daemon=True)
    thread.start()
    self.addCleanup(service.server_close)
    self.addCleanup(service.shutdown)
    url = f'http://127.0.0.1:{service.server_address[1]}/v1/github-issues/discover'
    payload = json.dumps({'repository_url': self.url, 'labels': [], 'state': 'open'}).encode()
    with self.assertRaises(error.HTTPError) as rejected:
        request.urlopen(request.Request(url, data=payload, headers={'Content-Type': 'application/json'}))
    self.assertEqual(rejected.exception.code, 401)
    rejected.exception.close()
    call = request.Request(url, data=payload, headers={'Content-Type': 'application/json', 'Authorization': 'Bearer ' + token})
    data = json.load(request.urlopen(call))
    self.assertEqual(len(data['issues']), 3)
    self.assertNotIn('DUMMY-PROVIDER', json.dumps(data))

def test_client_uncertain_delivery_and_bad_responses_are_not_rejections(self):
    import io
    from unittest.mock import patch
    from unittest.mock import MagicMock
    from urllib.error import HTTPError
    from laomedo.github.interface.issues.client import issue_request
    token_file = Path(self.temp.name) / 'dummy-control-token'
    token_file.write_text('x' * 40, encoding='utf-8')
    env = {'LAOMEDO_GITHUB_ISSUE_URL': 'http://127.0.0.1:8783', 'LAOMEDO_GITHUB_ISSUE_TOKEN_FILE': str(token_file)}
    payload = dict(repository_url=self.url, labels=[], state='open', issue_number=1, operation={'action': 'close', 'request_id': 'uncertain', 'grant_ref': 'grant'})
    with patch.dict('os.environ', env), patch('laomedo.github.interface.issues.client.request.build_opener') as opener:
        for failure in (TimeoutError('dummy-secret'), ConnectionResetError(), HTTPError('http://127.0.0.1', 503, 'dummy-secret', {}, io.BytesIO(b'{"error_category":"invalid_request"}'))):
            opener.return_value.open.side_effect = failure
            with self.assertRaisesRegex(ValueError, '^issue_write_unknown$'):
                issue_request('execute', **payload)
        opener.return_value.open.side_effect = None
        response = MagicMock()
        response.__enter__.return_value.read.return_value = b'[]'
        opener.return_value.open.return_value = response
        with self.assertRaisesRegex(ValueError, '^issue_write_unknown$'):
            issue_request('execute', **payload)
        opener.return_value.open.side_effect = HTTPError('http://127.0.0.1', 400, '', {}, io.BytesIO(b'{"error_category":"issue_grant_unavailable"}'))
        with self.assertRaisesRegex(ValueError, '^issue_grant_unavailable$'):
            issue_request('execute', **payload)

### PR 229 tests/test_module_cohesion.py::ModuleCohesionTests.test_production_classes_are_not_declared_inside_functions

def test_production_classes_are_not_declared_inside_functions(self):
    """Guard maintained implementation directories, not historical probes.

        Experiments, retained probes and inherited test/example doubles are
        deliberately outside this production guard; no historical-clean claim.
        """
    nested = []
    for directory in (ROOT / 'src', ROOT / 'components', ROOT / 'tools'):
        for path in directory.rglob('*.py'):
            tree = ast.parse(path.read_text(encoding='utf-8-sig'))
            parents = {child: node for node in ast.walk(tree) for child in ast.iter_child_nodes(node)}
            for node in ast.walk(tree):
                if not isinstance(node, ast.ClassDef):
                    continue
                ancestor = parents.get(node)
                while ancestor is not None:
                    if isinstance(ancestor, (ast.FunctionDef, ast.AsyncFunctionDef)):
                        nested.append(f'{path.relative_to(ROOT).as_posix()}:{node.lineno}:{node.name}')
                        break
                    ancestor = parents.get(ancestor)
    self.assertEqual(nested, [], 'Move implementation classes to module scope')
